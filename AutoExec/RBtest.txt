local MinecraftFonts = {
	-- Minecraft-style font mappings
	Title = Enum.Font.GothamBlack, -- Blocky for titles
	Header = Enum.Font.GothamBold, -- Bold headers
	Body = Enum.Font.Gotham, -- Regular text

	Minecraft = Enum.Font.SourceSansBold, -- Main Minecraft-like
}

-- Helper function to apply Minecraft style to text elements
local function applyMinecraftStyle(textElement, style, size)
	style = style or "Minecraft"

	-- Set font based on style
	textElement.Font = MinecraftFonts[style] or MinecraftFonts.Minecraft

	-- Adjust size for better Minecraft look
	if not size then
		if style == "Title" then
			textElement.TextSize = 20
		elseif style == "Header" then
			textElement.TextSize = 16
		elseif style == "Code" or style == "Pixel" then
			textElement.TextSize = 14
		else
			textElement.TextSize = 14
		end
	end

	-- Add text stroke for blocky Minecraft effect
	if not textElement:FindFirstChild("MinecraftStroke") then
		local stroke = Instance.new("UIStroke")
		stroke.Name = "MinecraftStroke"
		stroke.Color = Color3.new(0, 0, 0)
		stroke.Thickness = style == "Title" and 2 or 1
		stroke.Transparency = 0.5
		stroke.Parent = textElement
	end

	-- Minecraft text coloring
	if style == "Title" then
		textElement.TextColor3 = Colors.Text
	elseif style == "Header" then
		textElement.TextColor3 = Colors.Accent
	elseif style == "Code" then
		textElement.TextColor3 = Colors.Success
	else
		textElement.TextColor3 = Colors.Text
	end

	return textElement
end

-- Cache for active tweens to cancel before starting new ones
local activeTweens = setmetatable({}, { __mode = "k" })

local function playTween(object, tweenInfo, properties)
	-- Cancel existing tween for this object
	local existing = activeTweens[object]
	if existing then
		existing:Cancel()
	end

	local tween = Services.TweenService:Create(object, tweenInfo, properties)
	activeTweens[object] = tween
	tween:Play()

	-- Clean up reference when done
	tween.Completed:Connect(function()
		if activeTweens[object] == tween then
			activeTweens[object] = nil
		end
	end)

	return tween
end

local function colorToTable(c)
	return { r = math.floor(c.R * 255 + 0.5), g = math.floor(c.G * 255 + 0.5), b = math.floor(c.B * 255 + 0.5) }
end

local function tableToColor(t)
	return Color3.fromRGB(t.r or 255, t.g or 255, t.b or 255)
end

local function saveConfig()
	local dataToSave = {}

	-- Serialize each category
	for category, settings in pairs(ConfigData) do
		dataToSave[category] = {}
		for key, value in pairs(settings) do
			if typeof(value) == "Color3" then
				dataToSave[category][key] = colorToTable(value)
			else
				dataToSave[category][key] = value
			end
		end
	end

	local json = Services.HttpService:JSONEncode(dataToSave)
	local success = pcall(function()
		writefile(CONFIG_FILE, json)
	end)

	if success then
		print("✅ [Config] Saved to " .. CONFIG_FILE)
	else
		warn("❌ [Config] Failed to save")
	end

	return success
end

local configSaveVersion = 0
local configReady = false
local function scheduleConfigSave()
	if not configReady then
		return
	end
	configSaveVersion = configSaveVersion + 1
	local version = configSaveVersion
	task.delay(0.25, function()
		if version == configSaveVersion then
			saveConfig()
		end
	end)
end

local function loadConfig()
	if not isfile or not isfile(CONFIG_FILE) then
		print("ℹ️ [Config] No config file found")
		return false
	end

	local success, data = pcall(function()
		local content = readfile(CONFIG_FILE)
		return Services.HttpService:JSONDecode(content)
	end)

	if not success or not data then
		warn("❌ [Config] Failed to load")
		return false
	end

	local function mergeConfig(target, source)
		for key, value in pairs(source) do
			local current = target[key]
			if typeof(current) == "Color3" and type(value) == "table" then
				target[key] = tableToColor(value)
			elseif type(current) == "table" and type(value) == "table" then
				mergeConfig(current, value)
			elseif current ~= nil or key == "key" then
				target[key] = value
			end
		end
	end
	mergeConfig(ConfigData, data)
	print("✅ [Config] Loaded from " .. CONFIG_FILE)
	return true
end

	local self = setmetatable({}, UI)
	self.connections = {}
	self.visible = false

	self.activeTab = nil
	self.openDropdowns = {} -- legacy, kept for compat
	self.dropdownClosers = {} -- list of {isOpen, forceClose} per dropdown

	self.keybindEnforcers = {}

	self.hotkeyEntries = {}
	self.animationToken = 0
	self.isCapturingMenuKey = false

	self:_createScreenGui()
	self:_createHotkeyPanel()
	self:_createMainFrame()
	self:_createHeader()
	self:_createTabBar()
	self:_createContentArea()
	self:_setupToggle()

	local enforceElapsed = 0
	self:_connect(Services.RunService.Heartbeat, function(delta)
		enforceElapsed = enforceElapsed + delta
		if enforceElapsed < 0.25 then
			return
		end
		enforceElapsed = 0
		for _, enforce in ipairs(self.keybindEnforcers) do
			enforce()
		end
	end)

	return self
end

function UI:AnimateOpen()
	if not self.mainFrame then
		return
	end

	self.animationToken = self.animationToken + 1

	local centerPos = UDim2.new(0.5, -360, 0.5, -270)
	local startPos = UDim2.new(0.5, -360, 0.5, -258)

	self.mainFrame.Visible = true
	self.mainFrame.BackgroundTransparency = 1
	self.mainFrame.Position = startPos
	self.mainScale.Scale = 0.97

	local info = TweenInfo.new(0.24, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
	playTween(self.mainFrame, info, { BackgroundTransparency = 0, Position = centerPos })
	playTween(self.mainScale, info, { Scale = 1 })
	playTween(self.stroke, info, { Transparency = 0 })
	playTween(self.shadow, info, { ImageTransparency = 0.88 })
	playTween(self.shadowDark, info, { ImageTransparency = 0.7 })
end

function UI:AnimateClose(callback)
	if not self.mainFrame then
		if callback then
			callback()
		end
		return
	end

	self.animationToken = self.animationToken + 1
	local token = self.animationToken
	local endPos = UDim2.new(0.5, -360, 0.5, -282)
	local info = TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.In)

	playTween(self.mainFrame, info, { BackgroundTransparency = 1, Position = endPos })
	playTween(self.mainScale, info, { Scale = 0.97 })
	playTween(self.stroke, info, { Transparency = 1 })
	playTween(self.shadow, info, { ImageTransparency = 1 })
	playTween(self.shadowDark, info, { ImageTransparency = 1 })

	task.delay(0.17, function()
		if token == self.animationToken and callback then
			callback()
		end
	end)
end

function UI:_setupDrag()
	local dragging, dragStart, startPos = false, nil, nil

	self:_connect(self.mainFrame.InputBegan, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStart = input.Position
			startPos = self.mainFrame.Position
		end
	end)

	self:_connect(Services.UserInputService.InputChanged, function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			local delta = input.Position - dragStart
			self.mainFrame.Position =
				UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)

	self:_connect(Services.UserInputService.InputEnded, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)
end

function UI:SetActiveTab(name)
	for tabName, tabData in pairs(self.tabs) do
		local isActive = tabName == name
		tabData.content.Visible = isActive

		if isActive then
			-- Active: pill bg, colored text, slide indicator in
			playTween(tabData.button, TweenInfo_Fast, {
				BackgroundTransparency = 0.5,
				BackgroundColor3 = Colors.AccentDark,
			})
			playTween(tabData.text, TweenInfo_Fast, { TextColor3 = Colors.Text })
			playTween(tabData.icon, TweenInfo_Fast, { TextColor3 = Colors.AccentLight })
			playTween(tabData.indicator, TweenInfo_Medium, { Size = UDim2.new(0.7, 0, 0, 2) })
		else
			playTween(tabData.button, TweenInfo_Fast, {
				BackgroundTransparency = 1,
				BackgroundColor3 = Colors.BackgroundLighter,
			})
			playTween(tabData.text, TweenInfo_Fast, { TextColor3 = Colors.TextMuted })
			playTween(tabData.icon, TweenInfo_Fast, { TextColor3 = Colors.TextMuted })
			playTween(tabData.indicator, TweenInfo_Fast, { Size = UDim2.new(0, 0, 0, 2) })
		end
	end
	self.activeTab = name
end

function UI:CreateToggle(parent, text, default, callback, keybindConfig)
	local hotkeyId = {}
	local storedKeybind = keybindConfig and (keybindConfig.storage or keybindConfig)
	local toggle = Instance.new("Frame")
	toggle.Size = UDim2.new(1, 0, 0, 28)
	toggle.BackgroundColor3 = Colors.BackgroundLighter
	toggle.BackgroundTransparency = 1
	toggle.BorderSizePixel = 0

	local toggleCorner = Instance.new("UICorner")

	toggleCorner.Parent = toggle

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, -148, 1, 0)
	label.Position = UDim2.new(0, 6, 0, 0)
	label.BackgroundTransparency = 1
	label.Text = text
	label.TextColor3 = Colors.TextSecondary
	label.Font = MinecraftFonts.Minecraft
	label.TextSize = 12
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = toggle
	-- Don't applyMinecraftStyle - keeps text readable on dark background

	local switch = Instance.new("Frame")
	switch.Size = UDim2.new(0, 36, 0, 18)
	switch.Position = UDim2.new(1, -42, 0.5, -9)
	switch.BackgroundColor3 = Colors.ToggleOff
	switch.BorderSizePixel = 0
	switch.Parent = toggle

	local switchCorner = Instance.new("UICorner")
	switchCorner.CornerRadius = UDim.new(1, 0)
	switchCorner.Parent = switch

	local switchStroke = Instance.new("UIStroke")
	switchStroke.Color = Colors.Border
	switchStroke.Thickness = 1
	switchStroke.Parent = switch

	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, 14, 0, 14)
	knob.Position = UDim2.new(0, 2, 0.5, -7)
	knob.BackgroundColor3 = Colors.ToggleKnobOff
	knob.BorderSizePixel = 0
	knob.Parent = switch

	local knobCorner = Instance.new("UICorner")
	knobCorner.CornerRadius = UDim.new(1, 0)
	knobCorner.Parent = knob

	local state = nil
	local keybindMode = math.clamp(storedKeybind and storedKeybind.mode or 0, 0, 2)
	local currentKey = storedKeybind and storedKeybind.key or nil

	local function update(v, silent, force)
		if keybindMode == 2 and not v and not force then
			v = true
		end
		if state == v then
			return
		end
		state = v
		playTween(switch, TweenInfo_Fast, {
			BackgroundColor3 = v and Colors.Accent or Colors.ToggleOff,
		})
		playTween(switchStroke, TweenInfo_Fast, {
			Color = v and Colors.AccentDark or Colors.Border,
		})
		playTween(knob, TweenInfo_Fast, {
			Position = v and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7),
			BackgroundColor3 = v and Colors.Text or Colors.ToggleKnobOff,
		})
		playTween(label, TweenInfo_Fast, { TextColor3 = v and Colors.Text or Colors.TextSecondary })
		if self.hotkeyPanel then
			self:_updateHotkeyEntry(hotkeyId, text, currentKey, keybindMode, v)
		end
		if not silent and callback then
			callback(v)
		end
	end

	update(keybindMode == 2 or default or false, true, true)

	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 1, 0)
	btn.BackgroundTransparency = 1
	btn.Text = ""
	btn.ZIndex = 2
	btn.Parent = toggle

	self:_connect(btn.MouseEnter, function()
		if not state then
			playTween(toggle, TweenInfo_Fast, { BackgroundTransparency = 0.85 })
		end
	end)
	self:_connect(btn.MouseLeave, function()
		if not state then
			playTween(toggle, TweenInfo_Fast, { BackgroundTransparency = 1 })
		end
	end)

	self:_connect(btn.MouseButton1Click, function()
		if keybindMode == 2 then
			update(true, false, true)
			return
		end
		update(not state, false)
		if state then
			playTween(toggle, TweenInfo_Fast, { BackgroundTransparency = 1 })
		end
	end)

	local listening = false
	local listenConnection
	local listenToken = 0
	local menuOpen = false
	local modeButtons = {}
	local keyButton = Instance.new("TextButton")
	keyButton.Size = UDim2.new(0, 92, 0, 18)
	keyButton.Position = UDim2.new(1, -140, 0.5, -9)
	keyButton.BackgroundColor3 = Colors.BackgroundHover
	keyButton.BorderSizePixel = 0
	keyButton.TextColor3 = Colors.TextSecondary
	keyButton.Font = MinecraftFonts.Minecraft
	keyButton.TextSize = 10
	keyButton.ZIndex = 3
	keyButton.Parent = toggle

	local keyCorner = Instance.new("UICorner")
	keyCorner.CornerRadius = UDim.new(0, 4)
	keyCorner.Parent = keyButton

	local function syncKeybindConfig()
		if storedKeybind then
			storedKeybind.key = currentKey
			storedKeybind.mode = keybindMode
		end
		scheduleConfigSave()
	end

	local function updateKeyText()
		local modes = { "T", "H", "AO" }
		keyButton.Text = listening and "Ожидание"
			or currentKey and (currentKey .. " [" .. modes[keybindMode + 1] .. "]")
			or "None"
		keyButton.TextColor3 = listening and Colors.Warning or Colors.TextSecondary
		for mode, modeButton in pairs(modeButtons) do
			local active = mode == keybindMode
			modeButton.BackgroundColor3 = active and Colors.AccentLight or Colors.BackgroundLighter
			modeButton.TextColor3 = active and Colors.Text or Colors.TextSecondary
		end
		if self.hotkeyPanel then
			self:_updateHotkeyEntry(hotkeyId, text, currentKey, keybindMode, state == true)
		end
	end

	local menu = Instance.new("Frame")
	menu.Size = UDim2.new(0, 132, 0, 152)
	menu.BackgroundColor3 = Colors.BackgroundLight
	menu.BorderSizePixel = 0
	menu.Visible = false

	menuCorner.Parent = menu
	local menuStroke = Instance.new("UIStroke")

	menuStroke.Parent = menu
	local menuLayout = Instance.new("UIListLayout")

	local menuPadding = Instance.new("UIPadding")

	menuPadding.PaddingLeft = UDim.new(0, 4)
	menuPadding.PaddingRight = UDim.new(0, 4)
	menuPadding.Parent = menu

	local function closeMenu()
		menuOpen = false
		menu.Visible = false
		if self.keybindMenuCloser == closeMenu then
			self.keybindMenuCloser = nil
		end
	end

	local function stopListening()
		listening = false
		listenToken = listenToken + 1
		if listenConnection then
			listenConnection:Disconnect()
			listenConnection = nil
		end
		updateKeyText()
	end

	local function mouseNumberFromName(name)
		if not name then
			return nil
		end
		local number = string.match(name, "^MouseButton(%d+)$")
			or string.match(name, "^Mouse(%d+)$")
			or string.match(name, "^MB(%d+)$")
		if number then
			return number
		end
		if name == "BrowserBack" then
			return "4"
		end
		if name == "BrowserForward" then
			return "5"
		end
		return nil
	end

	local function mouseNumberFromInput(input)
		return mouseNumberFromName(input.UserInputType.Name)
			or mouseNumberFromName(input.KeyCode.Name)
			or mouseNumberFromName(tostring(input.UserInputType):match("%.([^%.]+)$"))
			or mouseNumberFromName(tostring(input.KeyCode):match("%.([^%.]+)$"))
	end

	local function startListening()
		stopListening()
		listening = true
		listenToken = listenToken + 1
		local token = listenToken
		local acceptAfter = os.clock() + 0.15
		updateKeyText()

		listenConnection = Services.UserInputService.InputBegan:Connect(function(input)
			if os.clock() < acceptAfter then
				return
			end
			local key
			local mouseNumber = mouseNumberFromInput(input)
			if mouseNumber then
				key = "MB" .. mouseNumber
			elseif input.KeyCode == Enum.KeyCode.Escape then
				stopListening()
				return
			elseif input.KeyCode ~= Enum.KeyCode.Unknown then
				local keyCodeName = input.KeyCode.Name
				local keyMouseNumber = mouseNumberFromName(keyCodeName)
				key = keyMouseNumber and ("MB" .. keyMouseNumber) or keyCodeName
			end

			if not key or token ~= listenToken then
				return
			end
			currentKey = key
			syncKeybindConfig()
			stopListening()
		end)
		table.insert(self.connections, listenConnection)

		task.delay(10, function()
			if token == listenToken then
				stopListening()
			end
		end)
	end

	local function setMode(mode)
		keybindMode = mode
		if keybindConfig and keybindConfig.onModeChanged then
			keybindConfig.onModeChanged(mode)
		end
		if mode == 1 then
			update(false, false, true)
		end
		if mode == 2 then
			update(true, false, true)
		end
		syncKeybindConfig()
		updateKeyText()
	end

	table.insert(self.keybindEnforcers, function()
		if keybindMode ~= 2 then
			return
		end
		if not state then
			update(true, false, true)
		elseif callback then
			callback(true)
		end
	end)

	local function addOption(name, action, mode)
		local option = Instance.new("TextButton")
		option.Size = UDim2.new(1, -8, 0, 27)
		option.BackgroundColor3 = Colors.BackgroundLighter
		option.BorderSizePixel = 0
		option.Text = name
		option.TextColor3 = Colors.Text
		option.Font = MinecraftFonts.Minecraft
		option.TextSize = 11
		option.ZIndex = 501
		option.Parent = menu
		local optionCorner = Instance.new("UICorner")
		optionCorner.CornerRadius = UDim.new(0, 4)
		optionCorner.Parent = option
		if mode ~= nil then
			modeButtons[mode] = option
		end
		self:_connect(option.MouseButton1Click, function()
			action()
			closeMenu()
		end)
	end

	addOption("Toggle", function()
		setMode(0)
		task.defer(startListening)
	end, 0)
	addOption("Hold", function()
		setMode(1)
		task.defer(startListening)
	end, 1)
	addOption("Always On", function()
		setMode(2)
	end, 2)
	local setKeyOption = Instance.new("TextButton")
	setKeyOption.Size = UDim2.new(1, -8, 0, 27)
	setKeyOption.BackgroundColor3 = Colors.BackgroundLighter
	setKeyOption.BorderSizePixel = 0
	setKeyOption.Text = "Set Key"
	setKeyOption.TextColor3 = Colors.Text
	setKeyOption.Font = MinecraftFonts.Minecraft
	setKeyOption.TextSize = 11
	setKeyOption.ZIndex = 501
	setKeyOption.Parent = menu
	local setKeyCorner = Instance.new("UICorner")
	setKeyCorner.CornerRadius = UDim.new(0, 4)
	setKeyCorner.Parent = setKeyOption
	self:_connect(setKeyOption.InputBegan, function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 then
			return
		end
		closeMenu()
		listening = true
		updateKeyText()
		task.defer(startListening)
	end)
	addOption("Clear", function()
		stopListening()
		currentKey = nil
		keybindMode = 0
		syncKeybindConfig()
		updateKeyText()
	end)

	local function openMenu()
		if self.keybindMenuCloser and self.keybindMenuCloser ~= closeMenu then
			self.keybindMenuCloser()
		end
		local position = keyButton.AbsolutePosition
		local viewport = Services.Workspace.CurrentCamera and Services.Workspace.CurrentCamera.ViewportSize
			or Vector2.new(1920, 1080)
		local x = math.clamp(position.X, 4, viewport.X - menu.Size.X.Offset - 4)
		local y = position.Y + keyButton.AbsoluteSize.Y + 2
		if y + menu.Size.Y.Offset > viewport.Y then
			y = position.Y - menu.Size.Y.Offset - 2
		end
		menu.Position = UDim2.fromOffset(x, math.max(4, y))
		menuOpen = true
		menu.Visible = true
		self.keybindMenuCloser = closeMenu
		updateKeyText()
	end

	self:_connect(keyButton.MouseButton1Click, function()
		if listening then
			stopListening()
		elseif currentKey then
			if menuOpen then
				closeMenu()
			else
				openMenu()
			end
		else
			startListening()
		end
	end)
	self:_connect(keyButton.InputBegan, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton2 then
			if menuOpen then
				closeMenu()
			else
				openMenu()
			end
		end
	end)

	local function matches(input)
		local expectedMouse = currentKey and mouseNumberFromName(currentKey)
		if expectedMouse then
			local actualMouse = mouseNumberFromInput(input)
			return actualMouse == expectedMouse
		end
		return input.KeyCode ~= Enum.KeyCode.Unknown and input.KeyCode.Name == currentKey
	end
	self:_connect(Services.UserInputService.InputBegan, function(input, processed)
		if listening then
			return
		end
		if not currentKey or keybindMode == 2 or not matches(input) then
			return
		end
		if processed and not mouseNumberFromName(currentKey) then
			return
		end
		if keybindMode == 1 then
			update(true, false, true)
		else
			update(not state, false, true)
		end
	end)
	self:_connect(Services.UserInputService.InputEnded, function(input)
		if keybindMode == 1 and currentKey and matches(input) then
			update(false, false, true)
		end
	end)
	self:_connect(Services.UserInputService.InputBegan, function(input)
		if not menuOpen or input.UserInputType ~= Enum.UserInputType.MouseButton1 then
			return
		end
		task.defer(function()
			if not menuOpen then
				return
			end
			local mouse = Services.UserInputService:GetMouseLocation()
			local menuPosition, menuSize = menu.AbsolutePosition, menu.AbsoluteSize
			local buttonPosition, buttonSize = keyButton.AbsolutePosition, keyButton.AbsoluteSize
			local inMenu = mouse.X >= menuPosition.X
				and mouse.X <= menuPosition.X + menuSize.X
				and mouse.Y >= menuPosition.Y
				and mouse.Y <= menuPosition.Y + menuSize.Y
			local inButton = mouse.X >= buttonPosition.X
				and mouse.X <= buttonPosition.X + buttonSize.X
				and mouse.Y >= buttonPosition.Y
				and mouse.Y <= buttonPosition.Y + buttonSize.Y
			if not inMenu and not inButton then
				closeMenu()
			end
		end)
	end)
	syncKeybindConfig()
	updateKeyText()

	local control = {
		Set = function(_, v)
			update(v, true, false)
		end,
		Get = function()
			return state
		end,
		SetKey = function(_, key)
			currentKey = key
			syncKeybindConfig()
			updateKeyText()
		end,
		GetKey = function()
			return currentKey
		end,
		SetMode = function(_, mode)
			setMode(math.clamp(mode, 0, 2))
		end,
		GetMode = function()
			return keybindMode
		end,
		RefreshKeybind = function()
			currentKey = storedKeybind and storedKeybind.key or nil
			setMode(math.clamp(storedKeybind and storedKeybind.mode or 0, 0, 2))
			updateKeyText()
		end,
	}
	table.insert(self.keybindControls, control)
	return control
end

function UI:ApplyHotkeyStyle(settings)
	if not self.hotkeyPanel then
		return
	end
	self.hotkeyScale.Scale = math.clamp((settings.hotkeysScale or 100) / 100, 0.6, 1.5)
	self.hotkeyPanel.BackgroundTransparency = math.clamp((settings.hotkeysTransparency or 12) / 100, 0, 1)
	self.hotkeyHeader.Visible = settings.hotkeysShowHeader ~= false
	self.hotkeyHeader.BackgroundColor3 = settings.hotkeysAccentColor or Colors.Accent
	self.hotkeyContainer.Position = UDim2.new(0, 0, 0, self.hotkeyHeader.Visible and 30 or 2)
	local viewport = Services.Workspace.CurrentCamera and Services.Workspace.CurrentCamera.ViewportSize
		or Vector2.new(1920, 1080)
	local x = math.clamp(
		tonumber(settings.hotkeysPositionX) or 18,
		0,
		math.max(0, viewport.X - self.hotkeyPanel.AbsoluteSize.X)
	)
	local y = math.clamp(
		tonumber(settings.hotkeysPositionY) or 400,
		0,
		math.max(0, viewport.Y - self.hotkeyPanel.AbsoluteSize.Y)
	)
	self.hotkeyPanel.Position = UDim2.fromOffset(x, y)
	self.hotkeyPanel:SetAttribute("EnabledByUser", settings.hotkeysEnabled ~= false)
	self:_refreshHotkeyVisibility()
end

function UI:_refreshHotkeyVisibility()
	local hasActiveHotkeys = false
	for _, entry in pairs(self.hotkeyEntries) do
		if entry.Parent and entry.Visible then
			hasActiveHotkeys = true
			break
		end
	end
	self.hotkeyPanel.Visible = self.hotkeyPanel:GetAttribute("EnabledByUser") ~= false and hasActiveHotkeys
end

	trackCorner.CornerRadius = UDim.new(1, 0)
	trackCorner.Parent = trackBg

	local fill = Instance.new("Frame")
	fill.Size = UDim2.new(math.clamp((default - min) / (max - min), 0, 1), 0, 1, 0)
	fill.BackgroundColor3 = Colors.Accent
	fill.BorderSizePixel = 0
	fill.Parent = trackBg

	local fillCorner = Instance.new("UICorner")
	fillCorner.CornerRadius = UDim.new(1, 0)
	fillCorner.Parent = fill

	-- Drag handle dot
	local handle = Instance.new("Frame")
	handle.Size = UDim2.new(0, 12, 0, 12)
	handle.AnchorPoint = Vector2.new(0.5, 0.5)
	handle.Position = UDim2.new(math.clamp((default - min) / (max - min), 0, 1), 0, 0.5, 0)
	handle.BackgroundColor3 = Colors.Text
	handle.BorderSizePixel = 0
	handle.ZIndex = 2
	handle.Parent = trackBg

	handleCorner.CornerRadius = UDim.new(1, 0)

	btn.Size = UDim2.new(1, 0, 1, 16)
	btn.Position = UDim2.new(0, 0, 0, -8)
	btn.BackgroundTransparency = 1
	btn.Text = ""
	btn.ZIndex = 3
	btn.Parent = trackBg

	local dragging = false

	local function updateSlider(value, notify)
		current = math.clamp(value, min, max)
		local fraction = (current - min) / (max - min)
		fill.Size = UDim2.new(fraction, 0, 1, 0)
		handle.Position = UDim2.new(fraction, 0, 0.5, 0)
		valueLabel.Text = tostring(current) .. suffix
		if notify and callback then
			callback(current)
		end
	end

	local function updateFromX(x)
		local fraction = math.clamp((x - trackBg.AbsolutePosition.X) / trackBg.AbsoluteSize.X, 0, 1)
		updateSlider(math.floor(min + (max - min) * fraction + 0.5), true)
	end

			dragging = true
			updateFromX(input.Position.X)
		end
	end)
	self:_connect(Services.UserInputService.InputChanged, function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			updateFromX(input.Position.X)
		end
	end)
	self:_connect(Services.UserInputService.InputEnded, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)

	return {
		Set = function(_, value)
			updateSlider(value, false)
		end,
		Get = function()
			return current
		end,
	}

		stroke.Color = Colors.BorderLight
		stroke.Thickness = 1
		stroke.Parent = popup

		local title = Instance.new("TextLabel")
		title.Size = UDim2.new(1, -44, 0, 30)
		title.Position = UDim2.fromOffset(12, 4)

		title.Parent = popup

		close.Size = UDim2.fromOffset(26, 26)
		close.Position = UDim2.new(1, -32, 0, 6)
		close.BackgroundColor3 = Colors.BackgroundLighter

function UI:CreateButton(parent, text, callback, style)
	style = style or "default"

	local baseColor = style == "danger" and Color3.fromRGB(50, 10, 10)
		or style == "success" and Color3.fromRGB(10, 50, 10)
		or Color3.fromRGB(20, 20, 20)
	local accentColor = style == "danger" and Colors.Danger or style == "success" and Colors.Success or Colors.Accent

	local button = Instance.new("TextButton")
	button.Size = UDim2.new(1, 0, 0, 30)
	button.BackgroundColor3 = baseColor
	button.BorderColor3 = accentColor
	button.BorderSizePixel = 2
	button.Text = ""
	button.AutoButtonColor = false
	button.Parent = parent

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, 0, 1, 0)
	label.BackgroundTransparency = 1
	label.Text = text
	label.TextColor3 = Colors.TextSecondary
	label.Font = MinecraftFonts.Header
	label.TextSize = 13
	label.Parent = button
	applyMinecraftStyle(label, "Header", 13)

	self:_connect(button.MouseEnter, function()
		playTween(button, TweenInfo_Fast, { BackgroundColor3 = accentColor, BorderColor3 = Colors.Text })
		playTween(label, TweenInfo_Fast, { TextColor3 = Colors.Text })
	end)
	self:_connect(button.MouseLeave, function()
		playTween(button, TweenInfo_Fast, { BackgroundColor3 = baseColor, BorderColor3 = accentColor })
		playTween(label, TweenInfo_Fast, { TextColor3 = Colors.TextSecondary })
	end)
	self:_connect(button.MouseButton1Down, function()
		playTween(button, TweenInfo_Fast, { Size = UDim2.new(1, 0, 0, 27) })
	end)
	self:_connect(button.MouseButton1Up, function()
		playTween(button, TweenInfo_Fast, { Size = UDim2.new(1, 0, 0, 30) })
	end)
	self:_connect(button.MouseButton1Click, callback)

	return button
end

function UI:SetVisible(visible)
	if self.visible == visible then
		print("[UI] SetVisible skipped - already", visible)
		return
	end
	self.visible = visible

	print("[UI] SetVisible called with:", visible)

	if visible then
		self.screenGui.Enabled = true
		self.mainFrame.Visible = true
		print("[UI] ScreenGui enabled, calling AnimateOpen")
		self:AnimateOpen()
	else
		print("[UI] Closing menu")
		self:AnimateClose(function()
			self.mainFrame.Visible = false
		end)
	end
end

function UI:Destroy()
	self.toastToken = (self.toastToken or 0) + 1
	for _, conn in ipairs(self.connections) do
		pcall(function()
			conn:Disconnect()
		end)
	end
	for object, tween in pairs(activeTweens) do
		pcall(function()
			if object:IsDescendantOf(self.screenGui) then
				tween:Cancel()
			end
		end)
	end
	for _, entry in pairs(self.hotkeyEntries) do
		pcall(function()
			entry:Destroy()
		end)
	end
	pcall(function()
		self.screenGui:Destroy()
	end)
end

function ESPObject:update(camera, localChar, settings)
	local char = self.player.Character
	if not char then
		self:hide()
		return
	end

	local hum = char:FindFirstChildOfClass("Humanoid")
	local root = char:FindFirstChild("HumanoidRootPart")
	local head = char:FindFirstChild("Head")

	if not hum or not root or not head or hum.Health <= 0 then
		self:hide()
		return
	end

	local lRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")
	if not lRoot then
		self:hide()
		return
	end

	local dist = (root.Position - lRoot.Position).Magnitude
	local sp, onScreen = camera:WorldToViewportPoint(root.Position)

	if not onScreen then
		self:hide()
		return
	end

	-- Calculate box dimensions
	local hsp = camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
	local fsp = camera:WorldToViewportPoint(root.Position - Vector3.new(0, 3, 0))

	local boxH = math.abs(hsp.Y - fsp.Y)
	local boxW = boxH * 0.6
	local boxX = sp.X - boxW / 2
	local boxY = hsp.Y

	-- Box
	if settings.boxEnabled then
		if settings.outlineEnabled then
			self.drawings.boxOutline.Size = Vector2.new(boxW, boxH)
			self.drawings.boxOutline.Position = Vector2.new(boxX, boxY)
			self.drawings.boxOutline.Color = settings.outlineColor or Color3.new(0, 0, 0)
			self.drawings.boxOutline.Visible = true
		end
		self.drawings.box.Size = Vector2.new(boxW, boxH)
		self.drawings.box.Position = Vector2.new(boxX, boxY)
		self.drawings.box.Color = settings.boxColor
		self.drawings.box.Visible = true
	else
		self.drawings.box.Visible = false
		self.drawings.boxOutline.Visible = false
	end

	-- Name
	if settings.nameEnabled then
		self.drawings.name.Text = self.player.DisplayName
		self.drawings.name.Position = Vector2.new(sp.X, boxY - 16)
		self.drawings.name.Color = settings.nameColor
		self.drawings.name.Visible = true
	else
		self.drawings.name.Visible = false
	end

	-- Distance
	if settings.distanceEnabled then
		self.drawings.distance.Text = string.format("[%dm]", math.floor(dist))
		self.drawings.distance.Position = Vector2.new(sp.X, fsp.Y + 2)
		self.drawings.distance.Color = settings.distanceColor
		self.drawings.distance.Visible = true
	else
		self.drawings.distance.Visible = false
	end

	-- Health bar
	if settings.healthBarEnabled then
		local hf = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
		local barW = 3
		local barX = boxX - barW - 4

		self.drawings.healthBg.Size = Vector2.new(barW, boxH)
		self.drawings.healthBg.Position = Vector2.new(barX, boxY)
		self.drawings.healthBg.Color = Color3.fromRGB(30, 30, 30)
		self.drawings.healthBg.Visible = true

		local fillH = boxH * hf
		self.drawings.healthFill.Size = Vector2.new(barW, fillH)
		self.drawings.healthFill.Position = Vector2.new(barX, boxY + boxH - fillH)
		-- Use custom health bar color or gradient based on health
		if settings.healthBarColor then
			self.drawings.healthFill.Color = settings.healthBarColor
		else
			self.drawings.healthFill.Color = Color3.fromRGB(255 - hf * 255, hf * 255, 0)
		end
		self.drawings.healthFill.Visible = true
	else
		self.drawings.healthBg.Visible = false
		self.drawings.healthFill.Visible = false
	end

	-- Snapline
	if settings.snaplineEnabled then
		local vp = camera.ViewportSize
		local fx, fy = vp.X / 2, vp.Y
		if settings.snaplineMode == "Top" then
			fy = 0
		elseif settings.snaplineMode == "Center" then
			fy = vp.Y / 2
		end

		self.drawings.snapline.From = Vector2.new(fx, fy)
		self.drawings.snapline.To = Vector2.new(sp.X, fsp.Y)
		self.drawings.snapline.Color = settings.snaplineColor
		self.drawings.snapline.Visible = true
	else
		self.drawings.snapline.Visible = false
	end

	-- Skeleton
	if settings.skeletonEnabled then
		self:_updateSkeleton(char, camera, settings.skeletonColor)
	else
		for _, line in ipairs(self.skeletonLines) do
			line.Visible = false
		end
	end

	-- Highlight
	if settings.highlightEnabled then
		if not self.highlight then
			self.highlight = Instance.new("Highlight")
			self.highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
			self.highlight.Parent = char
		end
		self.highlight.Adornee = char
		self.highlight.FillColor = settings.highlightFillColor
		self.highlight.OutlineColor = settings.highlightOutlineColor
		self.highlight.FillTransparency = settings.highlightFillTransparency
		self.highlight.OutlineTransparency = settings.highlightOutlineTransparency
		self.highlight.Enabled = true
	elseif self.highlight then
		self.highlight.Enabled = false
	end
end

function ESPObject:_updateSkeleton(char, camera, color)
	local r6Map = {
		Head = "Head",
		UpperTorso = "Torso",
		LowerTorso = "Torso",
		LeftUpperArm = "Left Arm",
		LeftLowerArm = "Left Arm",
		LeftHand = "Left Arm",
		RightUpperArm = "Right Arm",
		RightLowerArm = "Right Arm",
		RightHand = "Right Arm",
		LeftUpperLeg = "Left Leg",
		LeftLowerLeg = "Left Leg",
		LeftFoot = "Left Leg",
		RightUpperLeg = "Right Leg",
		RightLowerLeg = "Right Leg",
		RightFoot = "Right Leg",
	}

	local positions = {}
	for _, boneName in ipairs({
		"Head",
		"UpperTorso",
		"LowerTorso",
		"LeftUpperArm",
		"LeftLowerArm",
		"LeftHand",
		"RightUpperArm",
		"RightLowerArm",
		"RightHand",
		"LeftUpperLeg",
		"LeftLowerLeg",
		"LeftFoot",
		"RightUpperLeg",
		"RightLowerLeg",
		"RightFoot",
	}) do
		local part = char:FindFirstChild(boneName) or char:FindFirstChild(r6Map[boneName] or "")
		if part then
			local sp, onScreen = camera:WorldToViewportPoint(part.Position)
			if onScreen then
				positions[boneName] = Vector2.new(sp.X, sp.Y)
			end
		end
	end

	for i, conn in ipairs(BONE_CONNECTIONS) do
		local from, to = positions[conn[1]], positions[conn[2]]
		if from and to then
			self.skeletonLines[i].From = from
			self.skeletonLines[i].To = to
			self.skeletonLines[i].Color = color
			self.skeletonLines[i].Visible = true
		else
			self.skeletonLines[i].Visible = false
		end
	end
end

function ESPObject:hide()
	for _, drawing in pairs(self.drawings) do
		drawing.Visible = false
	end
	for _, line in ipairs(self.skeletonLines) do
		line.Visible = false
	end
	if self.highlight then
		self.highlight.Enabled = false
	end
end

function ESPObject:destroy()
	for _, drawing in pairs(self.drawings) do
		pcall(function()
			drawing:Remove()
		end)
	end
	for _, line in ipairs(self.skeletonLines) do
		pcall(function()
			line:Remove()
		end)
	end
	if self.highlight then
		pcall(function()
			self.highlight:Destroy()
		end)
	end
end

local MOUSE_KEYS = { MouseButton1 = true, MouseButton2 = true, MouseButton3 = true }

function AimbotSystem.new()
	local self = setmetatable({}, AimbotSystem)
	self.fovCircle = Drawing.new("Circle")
	self.fovCircle.Visible = false
	self.fovCircle.Thickness = 1
	self.fovCircle.Filled = false
	self.fovCircle.NumSides = 64
	self.fovCircle.Transparency = 0.8

	self.isActive = false
	self.currentTarget = nil
	self.triggerbotCooldown = 0

	return self
end

local TEAM_KEYS = {
	"Team",
	"TeamId",
	"TeamID",
	"TeamName",
	"TeamColor",
	"Faction",
	"FactionId",
	"FactionID",
	"Squad",
	"SquadId",
	"Side",
}

local function comparableTeamValue(value)
	local valueType = typeof(value)
	if valueType == "Instance" then
		return value:GetFullName()
	end
	if valueType == "BrickColor" then
		return value.Number
	end
	if valueType == "Color3" then
		return value:ToHex()
	end
	if valueType == "string" or valueType == "number" then
		return tostring(value)
	end
	return nil
end

local function readTeamValue(container, key)
	if not container then
		return nil
	end

	local attribute = container:GetAttribute(key)
	if attribute ~= nil then
		return comparableTeamValue(attribute)
	end

	local object = container:FindFirstChild(key)
	if object and object:IsA("ValueBase") then
		return comparableTeamValue(object.Value)
	end

	return nil
end

local function hasMatchingCustomTeam(playerA, playerB)
	local containersA = { playerA }
	local containersB = { playerB }
	if playerA.Character then
		table.insert(containersA, playerA.Character)
	end
	if playerB.Character then
		table.insert(containersB, playerB.Character)
	end
	local statsA = playerA:FindFirstChild("leaderstats")
	local statsB = playerB:FindFirstChild("leaderstats")
	if statsA then
		table.insert(containersA, statsA)
	end
	if statsB then
		table.insert(containersB, statsB)
	end

	for _, key in ipairs(TEAM_KEYS) do
		for _, containerA in ipairs(containersA) do
			local valueA = readTeamValue(containerA, key)
			if valueA ~= nil and valueA ~= "" and valueA ~= "0" then
				for _, containerB in ipairs(containersB) do
					local valueB = readTeamValue(containerB, key)
					if valueA == valueB then
						return true
					end
				end
			end
		end
	end

	return false
end

function AimbotSystem:isTeammate(player)
	if not player or player == LocalPlayer then
		return true
	end

	if LocalPlayer.Team ~= nil and player.Team ~= nil then
		return player.Team == LocalPlayer.Team
	end

	-- Some games keep players Neutral but still assign a meaningful TeamColor.
	local defaultTeamColor = BrickColor.new("Medium stone grey")
	if LocalPlayer.TeamColor == player.TeamColor and LocalPlayer.TeamColor ~= defaultTeamColor then
		return true
	end

	return hasMatchingCustomTeam(LocalPlayer, player)
end

local HITBOX_CANDIDATES = {
	Head = { "Head" },
	Neck = { "Head", "UpperTorso", "Torso" },
	UpperTorso = { "UpperTorso", "Torso" },
	LowerTorso = { "LowerTorso", "Torso", "HumanoidRootPart" },
	Torso = { "Torso", "UpperTorso", "LowerTorso" },
	LeftUpperArm = { "LeftUpperArm", "LeftLowerArm", "LeftHand", "Left Arm" },
	RightUpperArm = { "RightUpperArm", "RightLowerArm", "RightHand", "Right Arm" },
	LeftUpperLeg = { "LeftUpperLeg", "LeftLowerLeg", "LeftFoot", "Left Leg" },
	RightUpperLeg = { "RightUpperLeg", "RightLowerLeg", "RightFoot", "Right Leg" },
	HumanoidRootPart = { "HumanoidRootPart", "LowerTorso", "Torso" },
}

function AimbotSystem:getAimPart(player, bodyAim)
	if not player.Character then
		return nil
	end
	local char = player.Character

	local function findBodyPart(names)
		for _, name in ipairs(names) do
			local direct = char:FindFirstChild(name)
			if direct and direct:IsA("BasePart") then
				return direct
			end
		end

		for _, candidate in ipairs(char:GetDescendants()) do
			if candidate:IsA("BasePart") and table.find(names, candidate.Name) then
				if
					not candidate:FindFirstAncestorOfClass("Accessory")
					and not candidate:FindFirstAncestorOfClass("Tool")
				then
					return candidate
				end
			end
		end
		return nil
	end

	-- "Nearest" mode: pick the part whose screen position is closest to the mouse
	if bodyAim == "Nearest" then
		local cam = Services.Workspace.CurrentCamera
		if not cam then
			return nil
		end
		local mousePos = Services.UserInputService:GetMouseLocation()
		local best, bestDist = nil, math.huge

		for _, part in ipairs(char:GetDescendants()) do
			if
				part:IsA("BasePart")
				and not part:FindFirstAncestorOfClass("Accessory")
				and not part:FindFirstAncestorOfClass("Tool")
			then
				local sp, onScreen = cam:WorldToViewportPoint(part.Position)
				if onScreen and sp.Z > 0 then
					local dist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
					if dist < bestDist then
						bestDist = dist
						best = part
					end
				end
			end
		end
		return best
	end

	return findBodyPart(HITBOX_CANDIDATES[bodyAim] or { bodyAim })
end

function AimbotSystem:isVisible(part)
	local cam = Services.Workspace.CurrentCamera
	local origin = cam.CFrame.Position
	local dir = part.Position - origin

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	local exclude = {}
	if LocalPlayer.Character then
		table.insert(exclude, LocalPlayer.Character)
	end
	if part.Parent then
		table.insert(exclude, part.Parent)
	end
	params.FilterDescendantsInstances = exclude

	return Services.Workspace:Raycast(origin, dir, params) == nil
end

function AimbotSystem:getTarget(settings)
	if settings.stickyTarget and self.currentTarget then
		if settings.teamCheck and self:isTeammate(self.currentTarget) then
			self.currentTarget = nil
		end
	end

	if settings.stickyTarget and self.currentTarget then
		local char = self.currentTarget.Character
		if char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum and hum.Health > 0 then
				local aimPart = self:getAimPart(self.currentTarget, settings.bodyAim)
				if aimPart then
					local cam = Services.Workspace.CurrentCamera
					local sp, onScreen = cam:WorldToViewportPoint(aimPart.Position)
					if onScreen then
						local mousePos = Services.UserInputService:GetMouseLocation()
						local dist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
						if dist <= settings.fov then
							if not settings.wallCheck or self:isVisible(aimPart) then
								return self.currentTarget
							end
						end
					end
				end
			end
		end
	end

	local cam = Services.Workspace.CurrentCamera
	if not cam then
		return nil
	end
	local mousePos = Services.UserInputService:GetMouseLocation()
	local best, bestScore, bestScreenDist = nil, math.huge, math.huge
	local priority = settings.targetPriority or "Crosshair"
	local localRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

	for _, player in ipairs(Services.Players:GetPlayers()) do
		if player == LocalPlayer then
			continue
		end
		if settings.teamCheck and self:isTeammate(player) then
			continue
		end
		if not player.Character then
			continue
		end

		local hum = player.Character:FindFirstChildOfClass("Humanoid")
		if not hum or hum.Health <= 0 then
			continue
		end

		local aimPart = self:getAimPart(player, settings.bodyAim)
		if not aimPart then
			continue
		end

		local sp, onScreen = cam:WorldToViewportPoint(aimPart.Position)
		if not onScreen or sp.Z <= 0 then
			continue
		end

		local screenDist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
		if screenDist > settings.fov then
			continue
		end

		if settings.wallCheck and not self:isVisible(aimPart) then
			continue
		end

		local score = screenDist
		if priority == "Lowest Health" then
			score = hum.Health
		elseif priority == "Nearest Distance" then
			local targetRoot = player.Character:FindFirstChild("HumanoidRootPart") or aimPart
			local origin = localRoot and localRoot.Position or cam.CFrame.Position
			score = (targetRoot.Position - origin).Magnitude
		end

		if score < bestScore or (score == bestScore and screenDist < bestScreenDist) then
			bestScore = score
			bestScreenDist = screenDist
			best = player
		end
	end

	return best
end

function AimbotSystem:update(settings, delta)
	local cam = Services.Workspace.CurrentCamera
	local mousePos = Services.UserInputService:GetMouseLocation()

	-- FOV Circle
	if settings.enabled and settings.showFov then
		self.fovCircle.Position = mousePos
		self.fovCircle.Radius = settings.fov
		self.fovCircle.Color = settings.fovColor
		self.fovCircle.Visible = true
	else
		self.fovCircle.Visible = false
	end

	-- Triggerbot
	if settings.enabled and settings.triggerbotEnabled then
		self:processTriggerbot(settings, cam)
	end

	if settings.teamCheck and self.currentTarget and self:isTeammate(self.currentTarget) then
		self.currentTarget = nil
	end

	-- Aimbot
	if not settings.enabled or not self.isActive then
		return
	end

	local target = self:getTarget(settings)
	if not target then
		self.currentTarget = nil
		return
	end

	self.currentTarget = target
	local aimPart = self:getAimPart(target, settings.bodyAim)
	if not aimPart then
		return
	end

	local aimPos = aimPart.Position
	if settings.prediction > 0 then
		local vel = aimPart.AssemblyLinearVelocity or Vector3.zero
		aimPos = aimPos + vel * settings.prediction
	end

	local smooth = math.max(settings.smoothness, 1)
	local alpha = 1 / smooth

	if settings.lookAtView then
		local cf = CFrame.lookAt(cam.CFrame.Position, aimPos)
		cam.CFrame = cam.CFrame:Lerp(cf, alpha)
	else
		local sp = cam:WorldToViewportPoint(aimPos)
		local targetScreen = Vector2.new(sp.X, sp.Y)
		local diff = targetScreen - mousePos
		local moveBy = diff * alpha
		mousemoverel(moveBy.X, moveBy.Y)
	end
end

function UI:CreateKeybind(parent, text, default, callback)
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1, 0, 0, 34)
	row.BackgroundColor3 = Colors.BackgroundLighter
	row.BackgroundTransparency = 0.35
	row.BorderSizePixel = 0
	row.Parent = parent

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 6)
	corner.Parent = row

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, -126, 1, 0)
	label.Position = UDim2.fromOffset(8, 0)
	label.BackgroundTransparency = 1
	label.Text = text
	label.TextColor3 = Colors.TextSecondary
	label.Font = MinecraftFonts.Minecraft
	label.TextSize = 12
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = row

	local button = Instance.new("TextButton")
	button.Size = UDim2.fromOffset(108, 24)
	button.Position = UDim2.new(1, -114, 0.5, -12)
	button.BackgroundColor3 = Colors.AccentDark
	button.BorderSizePixel = 0
	button.TextColor3 = Colors.Text
	button.Font = MinecraftFonts.Code
	button.TextSize = 11
	button.AutoButtonColor = false
	button.Parent = row

	local buttonCorner = Instance.new("UICorner")
	buttonCorner.CornerRadius = UDim.new(0, 5)
	buttonCorner.Parent = button

	local buttonStroke = Instance.new("UIStroke")
	buttonStroke.Color = Colors.Accent
	buttonStroke.Transparency = 0.35
	buttonStroke.Thickness = 1
	buttonStroke.Parent = button

	local current = default or "F2"
	local listening = false

	local function render()
		button.Text = listening and "PRESS A KEY" or current
		button.TextColor3 = listening and Colors.Warning or Colors.Text
		buttonStroke.Color = listening and Colors.Warning or Colors.Accent
	end

	local function setKey(value, notify)
		if not Enum.KeyCode[value] or Enum.KeyCode[value] == Enum.KeyCode.Unknown then
			value = "F2"
		end
		current = value
		listening = false
		self.isCapturingMenuKey = false
		render()
		if notify and callback then
			callback(current)
		end
	end

	self:_connect(button.MouseEnter, function()
		playTween(button, TweenInfo_Fast, { BackgroundColor3 = Colors.Accent })
	end)
	self:_connect(button.MouseLeave, function()
		playTween(button, TweenInfo_Fast, { BackgroundColor3 = Colors.AccentDark })
	end)
	self:_connect(button.MouseButton1Click, function()
		listening = true
		self.isCapturingMenuKey = true
		render()
	end)
	self:_connect(Services.UserInputService.InputBegan, function(input)
		if not listening then
			return
		end
		if input.KeyCode == Enum.KeyCode.Escape then
			listening = false
			self.isCapturingMenuKey = false
			self.suppressMenuKeyUntil = os.clock() + 0.2
			render()
			return
		end
		if input.KeyCode == Enum.KeyCode.Backspace then
			self.suppressMenuKeyUntil = os.clock() + 0.2
			setKey("F2", true)
			return
		end
		if input.KeyCode == Enum.KeyCode.Unknown then
			return
		end
		self.suppressMenuKeyUntil = os.clock() + 0.2
		setKey(input.KeyCode.Name, true)
	end)

	render()
	return {
		Set = function(_, value)
			setKey(value, false)
		end,
		Get = function()
			return current
		end,
	}
end

function AimbotSystem:processTriggerbot(settings, camera)
	local now = tick()
	local delay = settings.triggerbotDelay / 1000
	if now - self.triggerbotCooldown < delay then
		return
	end

	local centerScreen = camera.ViewportSize / 2
	local tbFov = settings.triggerbotFov

	for _, player in ipairs(Services.Players:GetPlayers()) do
		if player == LocalPlayer then
			continue
		end
		if settings.teamCheck and self:isTeammate(player) then
			continue
		end
		if not player.Character then
			continue
		end

		local hum = player.Character:FindFirstChildOfClass("Humanoid")
		if not hum or hum.Health <= 0 then
			continue
		end

		for _, partName in ipairs({ "Head", "UpperTorso", "Torso", "HumanoidRootPart" }) do
			local part = player.Character:FindFirstChild(partName)
			if not part then
				continue
			end

			local sp, onScreen = camera:WorldToViewportPoint(part.Position)
			if not onScreen then
				continue
			end

			local dist = (Vector2.new(sp.X, sp.Y) - centerScreen).Magnitude
			if dist <= tbFov then
				if settings.wallCheck and not self:isVisible(part) then
					continue
				end

				self.triggerbotCooldown = now
				task.spawn(function()
					mouse1press()
					task.wait(0.03 + math.random() * 0.02)
					mouse1release()
				end)
				return
			end
		end
	end
end

function AimbotSystem:handleInput(input, gpe, settings, isBegin)
	if not settings.enabled then
		return
	end
	if isBegin and gpe then
		return
	end

	local key = KEY_MAP[settings.activationKey]
	if not key then
		return
	end

	local matched = false
	if MOUSE_KEYS[settings.activationKey] then
		matched = input.UserInputType == key
	else
		matched = input.KeyCode == key
	end

	if not matched then
		return
	end

	if isBegin then
		self.isActive = true
	else
		self.isActive = false
		if not settings.stickyTarget then
			self.currentTarget = nil
		end
	end
end

function AimbotSystem:destroy()
	pcall(function()
		self.fovCircle:Remove()
	end)
end

local MiscSystem = {}
MiscSystem.__index = MiscSystem

function MiscSystem.new()
	local self = setmetatable({}, MiscSystem)
	self.nightModeEnabled = false
	self.nightModeBrightness = 1
	self.originalLighting = nil
	return self
end

function MiscSystem:destroy()
	self:setNightMode(false)
end

function MiscSystem:setNightMode(enabled, settings)
	local lighting = Services.Lighting
	enabled = enabled == true
	settings = type(settings) == "table" and settings or ConfigData.misc
	local brightness = math.clamp(tonumber(settings.nightmodeBrightness) or 1, 0, 10)
	local ambientColor = settings.nightmodeAmbientColor or Color3.fromRGB(35, 35, 70)
	local outdoorColor = settings.nightmodeOutdoorColor or Color3.fromRGB(18, 18, 45)
	local fogColor = settings.nightmodeFogColor or Color3.fromRGB(20, 20, 50)
	if enabled then
		if not self.originalLighting then
			self.originalLighting = {
				Ambient = lighting.Ambient,
				OutdoorAmbient = lighting.OutdoorAmbient,
				Brightness = lighting.Brightness,
				ClockTime = lighting.ClockTime,
				FogColor = lighting.FogColor,
				FogEnd = lighting.FogEnd,
			}
		end
		self.nightModeEnabled = true
		self.nightModeBrightness = brightness
		lighting.Ambient = ambientColor
		lighting.OutdoorAmbient = outdoorColor
		lighting.Brightness = brightness
		lighting.ClockTime = 0
		lighting.FogColor = fogColor
		lighting.FogEnd = 1000
	elseif self.originalLighting then
		for property, value in pairs(self.originalLighting) do
			lighting[property] = value
		end
		self.originalLighting = nil
		self.nightModeEnabled = false
	end
end

function MiscSystem:updateWalkSpeed(speed, loop)
	if not loop then
		return
	end

	local char = LocalPlayer.Character
	if not char then
		return
	end

	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.WalkSpeed = speed
	end
end

function App.new()
	local self = setmetatable({}, App)

	self.ui = UI.new()
	self.espObjects = {}
	self.aimbot = AimbotSystem.new()
	self.misc = MiscSystem.new()
	self.uiElements = {} -- Store references to UI elements for config binding

	self:_buildUI()
	self:_setupESP()
	self:_setupAimbot()
	self:_setupMisc()
	self:_setupHotkeys()

	-- Load config after short delay
	task.spawn(function()
		task.wait(0.5)
		loadConfig()
		configReady = true
		self:_bindConfigToUI() -- Bind UI elements to ConfigData
		self:_applyConfig() -- Then apply values
	end)

	return self
end

function App:notify(title, message)
	print(string.format("[%s] %s", title, message))
	if self.ui then
		self.ui:Notify(title, message)
	end
end

function App:_setupESP()
	local wasEnabled = false
	local function onPlayerAdded(player)
		if player == LocalPlayer then
			return
		end
		self.espObjects[player] = ESPObject.new(player)
	end

	local function onPlayerRemoving(player)
		if self.espObjects[player] then
			self.espObjects[player]:destroy()
			self.espObjects[player] = nil
		end
	end

	for _, player in ipairs(Services.Players:GetPlayers()) do
		onPlayerAdded(player)
	end

	self.ui:_connect(Services.Players.PlayerAdded, onPlayerAdded)
	self.ui:_connect(Services.Players.PlayerRemoving, onPlayerRemoving)

	self.ui:_connect(Services.RunService.RenderStepped, function()
		if not ConfigData.esp.enabled then
			if wasEnabled then
				for _, obj in pairs(self.espObjects) do
					obj:hide()
				end
				wasEnabled = false
			end
			return
		end
		wasEnabled = true

		local cam = Services.Workspace.CurrentCamera
		local localChar = LocalPlayer.Character

		for player, obj in pairs(self.espObjects) do
			local isTeammate = ConfigData.esp.teamCheck and player.Team == LocalPlayer.Team
			if not isTeammate then
				obj:update(cam, localChar, ConfigData.esp)
			else
				obj:hide()
			end
		end
	end)
end

function App:_setupAimbot()
	self.ui:_connect(Services.RunService.RenderStepped, function(delta)
		self.aimbot:update(ConfigData.aimbot, delta)
	end)
end

function App:_setupMisc()
	local elapsed = 0
	self.ui:_connect(Services.RunService.Heartbeat, function(delta)
		elapsed = elapsed + delta
		if elapsed >= 0.25 then
			elapsed = 0
			if ConfigData.misc.nightmode and ConfigData.misc.nightmodePersist then
				self.misc:setNightMode(true, ConfigData.misc)
			end
		end
	end)
end

function App:_setupHotkeys()
	self.ui:_connect(Services.UserInputService.InputBegan, function(input, gpe)
		if self.ui.isCapturingMenuKey then
			if input.KeyCode == Enum.KeyCode.Unknown then
				return
			end
			self.ui.isCapturingMenuKey = false
			self.ui.suppressMenuKeyUntil = os.clock() + 0.2

			if input.KeyCode ~= Enum.KeyCode.Escape then
				ConfigData.misc.menuKey = input.KeyCode == Enum.KeyCode.Backspace and "F2" or input.KeyCode.Name
				scheduleConfigSave()
				self:notify("Menu Bind", "Menu key changed to " .. ConfigData.misc.menuKey)
			end
			if self.updateMenuBindButton then
				self.updateMenuBindButton(false)
			end
			return
		end
		if os.clock() < (self.ui.suppressMenuKeyUntil or 0) then
			return
		end
		if gpe then
			return
		end

		local menuKey = Enum.KeyCode[ConfigData.misc.menuKey or "F2"] or Enum.KeyCode.F2
		if input.KeyCode == menuKey then
			local isVisible = not self.ui.visible
			self.ui:SetVisible(isVisible)
		elseif input.KeyCode == Enum.KeyCode.F3 then
			saveConfig()
			self:notify("Config", "Saved!")
		elseif input.KeyCode == Enum.KeyCode.F4 then
			loadConfig()
			self:_applyConfig()
			self:notify("Config", "Loaded!")
		end
	end)
end

function App:destroy()
	if self.ui and self.ui.hotkeyPanel then
		ConfigData.misc.hotkeysPositionX = self.ui.hotkeyPanel.Position.X.Offset
		ConfigData.misc.hotkeysPositionY = self.ui.hotkeyPanel.Position.Y.Offset
	end
	if configReady then
		saveConfig()
	end
	self.ui:Destroy()
	self.aimbot:destroy()
	self.misc:destroy()

	for _, obj in pairs(self.espObjects) do
		obj:destroy()
	end
end
