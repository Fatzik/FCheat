local ConfigManager = {
    ConfigName = "BlatantConfig",
    ConfigFolder = "BlatantCat",
    ConfigPath = nil
}

function ConfigManager:GetPath()
    if self.ConfigPath then return self.ConfigPath end
    
    local paths = {
        "workspace/" .. self.ConfigFolder .. "/" .. self.ConfigName .. ".json",
        self.ConfigName .. ".json"
    }
    
    for _, path in ipairs(paths) do
        if isfile and isfile(path) then
            self.ConfigPath = path
            print("✅ Found config at: " .. path)
            return path
        end
    end
    
    self.ConfigPath = paths[1]
    print("📁 Config path set to: " .. self.ConfigPath)
    return self.ConfigPath
end

function ConfigManager:Save(data)
    local path = self:GetPath()
    local json = HttpService:JSONEncode(data)
    
    if isfolder then
        local folder = path:match("(.+)/[^/]+$")
        if folder and not isfolder(folder) then
            pcall(function() makefolder(folder) end)
        end
    end
    
    local success, err = pcall(function()
        writefile(path, json)
        print("💾 File saved: " .. path)
        print("📊 Size: " .. #json .. " bytes")
    end)
    
    if not success then
        print("❌ Save error: " .. tostring(err))
    end
    return success
end

function ConfigManager:Load()
    local path = self:GetPath()
    
    if not isfile or not isfile(path) then
        print("ℹ️ No config file at: " .. path)
        return nil
    end
    
    print("📂 Reading: " .. path)
    
    local success, data = pcall(function()
        local content = readfile(path)
        print("📊 File size: " .. #content .. " bytes")
        return HttpService:JSONDecode(content)
    end)
    
    if success and data then
        print("✅ Config decoded!")
        return data
    else
        print("❌ Failed to decode")
        return nil
    end
end

function UI:SaveConfig()
    print("💾 SAVING CONFIG...")
    print("=" .. string.rep("-", 40))
    
    local config = {
        espSettings = {},
        aimbotSettings = {},
        miscSettings = {},
        espEnabled = self._espEnabled,
        version = "1.0",
        timestamp = os.time()
    }
    
    -- ESP
    for key, value in pairs(self._espSettings) do
        if typeof(value) == "Color3" then
            config.espSettings[key] = {R = value.R, G = value.G, B = value.B}
        else
            config.espSettings[key] = value
        end
    end
    
    -- AIMBOT (ВКЛЮЧАЯ SMOOTHNESS)
    print("📝 Aimbot settings:")
    for key, value in pairs(self._aimbotSettings) do
        if typeof(value) == "Color3" then
            config.aimbotSettings[key] = {R = value.R, G = value.G, B = value.B}
        else
            config.aimbotSettings[key] = value
            print("  " .. key .. " = " .. tostring(value))
        end
    end
    
    -- MISC
    for key, value in pairs(self._miscSettings) do
        config.miscSettings[key] = value
    end
    
    print("=" .. string.rep("-", 40))
    
    local success = ConfigManager:Save(config)
    if success then
        self._configLoaded = true
        print("✅ CONFIG SAVED SUCCESSFULLY!")
        print("📊 Smoothness = " .. tostring(self._aimbotSettings.Smoothness))
        self:ShowNotification("✅ Config Saved!", "Smoothness: " .. self._aimbotSettings.Smoothness)
    else
        print("❌ FAILED TO SAVE CONFIG!")
        self:ShowNotification("❌ Save Failed!", "Check console for errors")
    end
    return success
end

function UI:LoadConfig()
    print("📂 LOADING CONFIG...")
    print("=" .. string.rep("-", 40))
    
    local config = ConfigManager:Load()
    if not config then
        print("ℹ️ No config found, using defaults")
        self:ShowNotification("ℹ️ No Config", "Using default settings")
        return false
    end
    
    local loaded = false
    
    -- ESP
    if config.espSettings then
        for key, value in pairs(config.espSettings) do
            if type(value) == "table" and value.R and value.G and value.B then
                self._espSettings[key] = Color3.new(value.R, value.G, value.B)
                loaded = true
            elseif self._espSettings[key] ~= nil then
                self._espSettings[key] = value
                loaded = true
            end
        end
    end
    
    -- AIMBOT (ВКЛЮЧАЯ SMOOTHNESS)
    if config.aimbotSettings then
        print("📥 Loaded aimbot settings:")
        for key, value in pairs(config.aimbotSettings) do
            if type(value) == "table" and value.R and value.G and value.B then
                self._aimbotSettings[key] = Color3.new(value.R, value.G, value.B)
                loaded = true
            elseif self._aimbotSettings[key] ~= nil then
                self._aimbotSettings[key] = value
                print("  " .. key .. " = " .. tostring(value))
                loaded = true
            end
        end
    end
    
    -- MISC
    if config.miscSettings then
        for key, value in pairs(config.miscSettings) do
            if self._miscSettings[key] ~= nil then
                self._miscSettings[key] = value
                loaded = true
            end
        end
    end
    
    if config.espEnabled ~= nil then
        self._espEnabled = config.espEnabled
        loaded = true
    end
    
    if loaded then
        self._configLoaded = true
        print("=" .. string.rep("-", 40))
        print("✅ CONFIG LOADED SUCCESSFULLY!")
        print("📊 Smoothness = " .. tostring(self._aimbotSettings.Smoothness))
        self:ShowNotification("✅ Config Loaded!", "Smoothness: " .. self._aimbotSettings.Smoothness)
        
        if self._miscSettings.FlyEnabled then
            self:ToggleFly(true)
        end
        return true
    else
        print("❌ No settings were loaded!")
        return false
    end
end

function UI:_buildDrag()
    local dragging = false
    local dragStart, startPos

    self:_connect(self._mainFrame.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = self._mainFrame.Position
            local c
            c = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if c then c:Disconnect() end
                end
            end)
        end
    end)

    self:_connect(UserInputService.InputChanged, function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement and dragging then
            local delta = input.Position - dragStart
            self._mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

        content.BackgroundTransparency = 1

    self:_buildAimbotTab()
    self:_buildMiscTab()

function UI:_switchTab(name)
    for tabName, content in pairs(self._tabContents) do
        content.Visible = (tabName == name)
        self._tabButtons[tabName].TextColor3 = (tabName == name) and Color3.fromRGB(0, 180, 180) or Color3.fromRGB(120, 130, 140)
    end
    self._activeTab = name
end

    local sliding = false

    local function update(input)
        local rel = math.clamp((input.Position.X - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * rel + 0.5)
        fillBar.Size = UDim2.new((val - min) / (max - min), 0, 1, 0)
        valLabel.Text = tostring(val) .. suffix
        if callback then callback(val) end
    end

    self:_connect(sliderBtn.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            sliding = true
            update(input)
        end
    end)

    self:_connect(UserInputService.InputChanged, function(input)
        if sliding and input.UserInputType == Enum.UserInputType.MouseMovement then
            update(input)
        end
    end)

    self:_connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            sliding = false
        end
    end)

    return frame
end

function UI:_setupToggle()
    self:_connect(UserInputService.InputBegan, function(input, gpe)
        if gpe then return end
        if input.KeyCode == Enum.KeyCode.F2 then
            self._visible = not self._visible
            self._screenGui.Enabled = self._visible
        end
        
        if input.KeyCode == Enum.KeyCode.F3 then
            self:SaveConfig()
        elseif input.KeyCode == Enum.KeyCode.F4 then
            self:LoadConfig()
        end
    end)
end

-- MISC
-- ============================================
function UI:ToggleFly(enabled)
    if enabled then
        self._flyState.active = true
        pcall(function()
            local char = LocalPlayer.Character
            if not char then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            local head = char:WaitForChild("Head")
            if not hum or not head then return end
            
            if self._flyState.cfLoop then
                self._flyState.cfLoop:Disconnect()
                self._flyState.cfLoop = nil
            end
            
            hum.PlatformStand = true
            head.Anchored = true
            
            self._flyState.cfLoop = RunService.Heartbeat:Connect(function(deltaTime)
                pcall(function()
                    local c = LocalPlayer.Character
                    if not c then return end
                    local h = c:FindFirstChildOfClass("Humanoid")
                    local hd = c:FindFirstChild("Head")
                    if not h or not hd then return end
                    local speed = self._miscSettings.FlySpeed
                    local moveDirection = h.MoveDirection * (speed * deltaTime)
                    local headCFrame = hd.CFrame
                    local camera = Workspace.CurrentCamera
                    local cameraCFrame = camera.CFrame
                    local cameraOffset = headCFrame:ToObjectSpace(cameraCFrame).Position
                    cameraCFrame = cameraCFrame * CFrame.new(-cameraOffset.X, -cameraOffset.Y, -cameraOffset.Z + 1)
                    local cameraPosition = cameraCFrame.Position
                    local headPosition = headCFrame.Position
                    local objectSpaceVelocity = CFrame.new(cameraPosition, Vector3.new(headPosition.X, cameraPosition.Y, headPosition.Z)):VectorToObjectSpace(moveDirection)
                    hd.CFrame = CFrame.new(headPosition) * (cameraCFrame - cameraPosition) * CFrame.new(objectSpaceVelocity)
                end)
            end)
            table.insert(self._connections, self._flyState.cfLoop)
        end)
    else
        self._flyState.active = false
        if self._flyState.cfLoop then
            self._flyState.cfLoop:Disconnect()
            self._flyState.cfLoop = nil
        end
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local head = char:FindFirstChild("Head")
                if head then head.Anchored = false end
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum then hum.PlatformStand = false end
            end
        end)
    end
end

function UI:destroy()
    for _, c in ipairs(self._connections) do pcall(function() c:Disconnect() end) end
    for _, o in pairs(self._espObjects) do o:destroy() end
    if self._fovCircle then pcall(function() self._fovCircle:Remove() end) end
    if self._flyState.cfLoop then pcall(function() self._flyState.cfLoop:Disconnect() end) end
    for _, pf in ipairs(self._openColorPickers) do
        pcall(function() pf:Destroy() end)
    end
    pcall(function() self._screenGui:Destroy() end)
    self._connections = {}
    self._espObjects = {}
end
